export default {
  async fetch(request) {
    const corsHeaders = {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Methods": "GET, HEAD, POST, OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type",
    };

    if (request.method === "OPTIONS") {
      return new Response(null, { headers: corsHeaders });
    }

    try {
      const STATION = "LNS";
      const API = "https://data.rcc-acis.org/StnData";

      // =====================================================
      // DATE ENGINE (SOURCE OF TRUTH)
      // =====================================================
      const pad = (n) => String(n).padStart(2, "0");

      const fmt = (d) =>
        `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;

      const today = new Date(
        new Date().toLocaleString("en-US", { timeZone: "America/New_York" })
      );

      const yesterday = new Date(today);
      yesterday.setDate(today.getDate() - 1);

      const todayStr = fmt(today);
      const ydayStr = fmt(yesterday);

      // =====================================================
      // YTD RANGE (Jan 1 → Yesterday)
      // =====================================================
      const ytdStartYear =
        yesterday.getFullYear();

      const ytdStart = `${ytdStartYear}-01-01`;
      const ytdEnd = ydayStr;

      // =====================================================
      // SNOW SEASON RANGE (Jul 1 → Jun 30)
      // =====================================================
      const snowStartYear =
        today.getMonth() >= 6
          ? today.getFullYear()
          : today.getFullYear() - 1;

      const snowSeasonStart = `${snowStartYear}-07-01`;
      const snowSeasonEnd = `${snowStartYear + 1}-06-30`;

      // =====================================================
      // ACIS FETCH HELPER
      // =====================================================
      const fetchAcis = async (params) => {
        const res = await fetch(
          `${API}?params=${encodeURIComponent(JSON.stringify(params))}`
        );
        return res.json();
      };

      // =====================================================
      // FETCH DATASETS
      // =====================================================
      const [ytdRaw, todayRaw, snowRaw] = await Promise.all([
        fetchAcis({
          sid: STATION,
          sdate: ytdStart,
          edate: ytdEnd,
          elems: ["maxt", "mint", "pcpn", "snow"],
        }),

        fetchAcis({
          sid: STATION,
          sdate: todayStr,
          edate: todayStr,
          elems: [
            { name: "maxt", normal: "1" },
            { name: "mint", normal: "1" },
            { name: "pcpn", normal: "1" },
          ],
        }),

        fetchAcis({
          sid: STATION,
          sdate: snowSeasonStart,
          edate: snowSeasonEnd,
          elems: ["snow"],
        }),
      ]);

      // =====================================================
      // SAFE PARSER
      // =====================================================
      const num = (v) => {
        if (v === "M" || v === "T" || v == null) return 0;
        const n = Number(v);
        return Number.isFinite(n) ? n : 0;
      };

      // =====================================================
      // YTD PROCESSING
      // =====================================================
      let ytdRain = 0;
      let ytdNormRain = 0;
      let monthlyRows = [];

      const monthKey = todayStr.slice(0, 7);

      for (const r of ytdRaw?.data || []) {
        const [date, maxt, mint, pcpn, snow, maxtN, mintN, pcpnN] = r;

        ytdRain += num(pcpn);
        ytdNormRain += num(pcpnN);

        if (date?.slice(0, 7) === monthKey) {
          monthlyRows.push([date, maxt, mint, pcpn]);
        }
      }

      const ytdRainDeparture = ytdRain - ytdNormRain;

      // =====================================================
      // TODAY NORMALS
      // =====================================================
      const t = todayRaw?.data?.[0] || [];

      const todayNormals = {
        normalHigh: num(t[1]),
        normalLow: num(t[2]),
        normalRain: num(t[3]),
      };

      // =====================================================
      // SNOW SEASON PROCESSING
      // =====================================================
      let seasonalTotal = 0;
      let snowDays = 0;
      let maxSnow = 0;
      let maxSnowDate = "";

      for (const r of snowRaw?.data || []) {
        const [date, snow] = r;
        const s = num(snow);

        if (s > 0) {
          seasonalTotal += s;
          snowDays++;

          if (s > maxSnow) {
            maxSnow = s;
            maxSnowDate = date;
          }
        }
      }

      // =====================================================
      // RESPONSE (CLEAN API SURFACE)
      // =====================================================
      return new Response(
        JSON.stringify({
          station: STATION,
          date: todayStr,

          monthlyRows,

          ytdRain,
          ytdRainDeparture,

          todayNormals,

          snow: {
            seasonStart: snowSeasonStart,
            seasonEnd: snowSeasonEnd,
            total: seasonalTotal,
            days: snowDays,
            max: maxSnow,
            maxDate: maxSnowDate,
          },
        }),
        {
          headers: {
            ...corsHeaders,
            "Content-Type": "application/json",
          },
        }
      );

    } catch (err) {
      return new Response(
        JSON.stringify({ error: err.message }),
        {
          status: 500,
          headers: {
            "Content-Type": "application/json",
            ...corsHeaders,
          },
        }
      );
    }
  },
};
